# Harness runs for the Jolt harness workflow (.github/workflows/jolt-harness.yml), on every runner in its matrix.
# One run per line: a name (letters, digits, - and _), a colon, then the harness arguments. Each run writes its
# per-heartbeat traces and summary.csv under <name>/ in the uploaded artifact. Lines starting with # are skipped.

# The scenarios of the tests that came out differently on Linux and Windows before the natives were built with
# CROSS_PLATFORM_DETERMINISTIC.
crash-headon-11: --scenario crash-headon --rate 11
crash-box-11: --scenario crash-box --rate 11
crash-all: --scenario crash-wall,crash-box,crash-headon,crash-drop --rate 11,45
park-drive-45-legacy: --scenario park-drive --rate 45 --jolt VehiclePresets=legacy
testcar-down-5: --scenario testcar-down --slope 5 --rate all

# Crash sweep: small changes in the arrival speed, sideways offset and angle, at 11 and 45 Hz. Penetration is the
# boxes' depth into each other; crash_rise how far either object rose in the second after the impact.
crash-sweep: --scenario crash-wall,crash-box,crash-headon,crash-drop --rate 11,45 --crash-speed 0.9,0.95,1,1.05,1.1 --crash-offset -0.2,0,0.2 --crash-angle -1,0,1

# Script ray casts: what one costs with its cost bounded (raycast-cost) and made as before, every hit along the ray
# (raycast-cost-all); the summary's ray_us_per_cast. Pushes on an avatar, a parked car and a sleeping box.
raycast-cost: --scenario raycast-cost,raycast-cost-all --rate 11 --physics-rate 0,45
push: --scenario push-avatar,push-flood,push-car,push-box --rate 11 --physics-rate 0,45

# Arriving avatars: one arriving on a fixed platform 3 m above the ground stays on it; one arriving 3 m above the
# platform, not flying, falls and lands on it. end_z is where each stands.
avatar-arrive: --scenario avatar-platform,avatar-platform-drop --rate 11,45 --physics-rate 0,45

# Ten 0.5 m boxes stacked on the ground at [Jolt] VelocityIterations 10, 20 and 40: does the tower stand and fall
# asleep, and when (the console's "part" lines; every box's trace in tower-10-*-parts.csv).
tower-10-v10: --scenario tower-10 --rate 11 --physics-rate 0,22.5,45,90 --jolt VelocityIterations=10
tower-10-v20: --scenario tower-10 --rate 11 --physics-rate 0,22.5,45,90 --jolt VelocityIterations=20
tower-10-v40: --scenario tower-10 --rate 11 --physics-rate 0,22.5,45,90 --jolt VelocityIterations=40
tower-10-v40-cs3: --scenario tower-10 --rate 11 --physics-rate 22.5 --jolt VelocityIterations=40 --jolt PhysicsStepCollisionSteps=3
tower-10-v20-cs3: --scenario tower-10 --rate 11 --physics-rate 22.5 --jolt VelocityIterations=20 --jolt PhysicsStepCollisionSteps=3
